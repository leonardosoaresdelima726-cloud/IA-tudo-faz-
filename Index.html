// Alternar Abas

function mudarAba(nomeAba){

  document.querySelectorAll('[id^="sec-"]').forEach(s=>s.classList.add('hidden'));

  document.getElementById(`sec-${nomeAba}`).classList.remove('hidden');

 

  document.querySelectorAll('[id^="aba-"]').forEach(b=>{

    b.classList.remove('tab-active');

    b.classList.add('bg-white/5','hover:bg-white/10');

  });

  document.getElementById(`aba-${nomeAba}`).classList.add('tab-active');

  document.getElementById(`aba-${nomeAba}`).classList.remove('bg-white/5','hover:bg-white/10');

}

// Sistema de Chat

function adicionarMensagem(remetente,texto,isUsuario){

  const caixa=document.getElementById('caixaMensagens');

  const div=document.createElement('div');

  div.className='flex gap-3'+(isUsuario?' flex-row-reverse':'');

  div.innerHTML=`

    <div class="w-8 h-8 rounded-full ${isUsuario?'bg-green-500':'bg-gradient-to-br from-primary-500 to-magic-500'} flex items-center justify-center flex-shrink-0">

      <i class="fa ${isUsuario?'fa-user':'fa-robot'} text-xs"></i>

    </div>

    <div class="message-bubble ${isUsuario?'bg-primary-600/30 rounded-2xl rounded-tr-none':'glass rounded-2xl rounded-tl-none'} p-4">

      <p>${texto}</p>

    </div>

  `;

  caixa.appendChild(div);

  caixa.scrollTop=caixa.scrollHeight;

}

async function enviarMensagem(){

  const input=document.getElementById('entradaMensagem');

  const texto=input.value.trim();

  if(!texto)return;

 

  adicionarMensagem('usuario',texto,true);

  input.value='';

 

  // Mostrar digitando

  const caixa=document.getElementById('caixaMensagens');

  const digitando=document.createElement('div');

  digitando.id='digitando';

  digitando.className='flex gap-3';

  digitando.innerHTML=`

    <div class="w-8 h-8 rounded-full bg-gradient-to-br from-primary-500 to-magic-500 flex items-center justify-center flex-shrink-0">

      <i class="fa fa-robot text-xs"></i>

    </div>

    <div class="message-bubble glass rounded-2xl rounded-tl-none p-4">

      <span class="typing-dot"></span>

      <span class="typing-dot"></span>

      <span class="typing-dot"></span>

      <span class="text-gray-400 ml-2">Pensando...</span>

    </div>

  `;

  caixa.appendChild(digitando);

  caixa.scrollTop=caixa.scrollHeight;

 

  await new Promise(r=>setTimeout(r,1500));

  digitando.remove();

 

  // Resposta inteligente

  let resposta=gerarResposta(texto);

  adicionarMensagem('ia',resposta,false);

}

  }

  if(p.includes('vídeo')||p.includes('video')||p.includes('filme')||p.includes('anúncio')){

    return "Boa! 🎬 Vamos criar um vídeo! Vá na aba **Vídeo**, descreva a cena, o estilo e a duração. Eu te entrego o roteiro completo pronto para gravar! 🎥✨";

  if(p.includes('site')||p.includes('aplicativo')||p.includes('app')||p.includes('página')){

    return "Excelente! 🌐 Vou criar seu projeto! Vá na aba **Site/App**, preencha nome, descrição e cor. Eu gero o código completo pronto para usar e publicar! 🚀";

  if(p.includes('código')||p.includes('programa')||p.includes('script')||p.includes('função')){

  }

  if(p.includes('olá')||p.includes('oi')||p.includes('bom dia')||p.includes('boa tarde')){

    return "Olá! 🤖✨ Que alegria! Estou aqui para te ajudar a criar TUDO! Posso gerar imagens, vídeos, sites, aplicativos, códigos e muito mais! É só pedir! O que vamos fazer primeiro? 😊";

  }

  if(p.includes('quem é você')||p.includes('o que você faz')){

    return "Eu sou a **IA TudoFaz**! 🤖✨ Seu assistente universal! Crio imagens, vídeos, sites, aplicativos, códigos, textos e projetos do zero! Tudo aqui, num só lugar! É só me dizer o que precisa! 💪🚀";

  setTimeout(()=>{

    document.getElementById('previewImagem').innerHTML=`

      <div class="text-center p-4">
